服从各位大神的指令，我来把毕生（bushi）所学的C语言基础知识记录下来，方便自己复习（完成任务）。\

if语句用于条件判断，格式为：
if(条件) {
条件为真时执行的代码
}
else {
条件为假时执行的代码
}
just：避雷鄙人在判断a是不是等于5时，写成了if(a=5)，导致a被赋值为5，条件永远为真。正确的写法应该是if(a==5)。
else：就近原则（学英语就记住这个了），else语句会与最近的if语句匹配。
eg：if(i > 1){
if(j > 2)
printf(“i>1且j>2\n”);
}
else{
printf(“i<=1\n”);//i <= 1时执行
}
for循环用于重复执行代码块，格式为：
for(初始化; 条件; 更新) {
//循环体代码,符合条件循环执行
}
大坑：;别写成,了，第一次做for循环题目时，写成了for(i=0, i<10, i++)，导致循环体只执行了一次。正确的写法应该是for(i=0; i<10; i++)。
问了ai才知道要写;，这波der了。。。
括号里定义i = 0后，后面不能用i <= n 了，写成i < n才行，否则会多执行一次循环体。还有这个i出了for就不好使了，别问我咋知道的，问就是室友发现的（bushi）。

while循环用于在条件为真时重复执行代码块，格式为：
while(条件) {
//循环体代码,符合条件循环执行
//tip:记得更新变量，要不就堕入死循环了（i++）
}
int i = 1;
while(i <= 5 ){
printf(“%d”,i);
//不写i++就炸钢了
i++；
}
不知道循环多少次就用while,没毛病。

switch语句：精准打击
switch (变量) {
case 常量1:
// 执行语句
break;
case 常量2:
// 执行语句
break;
…
default:
// 所有 case 都不匹配时，执行这里
}
switch比if更精准，if是判断范围的。
还有就是每个都要写break，没写就卡了。

函数：
我的理解就是自己写一段像printf scanf这类然后在后续代码里直接用，省的用多次时候费劲巴力写一大堆。
返回值类型 函数名(参数列表) {
// 函数体：具体执行的代码
return 返回值; // 如果有返回值类型，就要写 return
}
eg：
#include <stdio.h>

// 返回值是 int，参数是两个 int
int add(int a, int b) {
return a + b; // 把结果交回去
}

int main() {
int x = 3, y = 5;
int result = add(x, y);
printf(“结果是：%d\n”, result); // 输出 8
return 0;
}
就是自己写一段便于自己直接使用的代码，后面可以多次使用。

数组：
数组就是一个有2，3，4，5，个盘的硬盘一样，int arr [5];就是分了5个盘。
原来的int i只能赋值一个，就算可以也要int i,o,j,h,g,f,d,e,w,q,a,z,写一大堆（不嫌费事就写hhh）int arr [n];就解决了。
#include <stdio.h>

int main() {
int arr[5];
printf(“请输入 5 个整数：\n”);
// 循环存入数据（下标从 0 到 4）
for (int i = 0; i < 5; i++) {//刚才提到的 <= 和 < 区别
scanf(“%d”, &arr[i]); //加 &，取地址
}
printf(“你输入的数字是：”);
for (int i = 0; i < 5; i++) {
printf(“%d “, arr[i]);
}
return 0;
}
这是学习的时候ai给的例子，存入数据和遍历输出。

难中之难 指针。。。
int p;
ai说指针是个装另一个盒子的门牌号的盒子，我觉得这个比喻挺合理挺对的。
int a = 10;
int p = &a;
a和p一伙用printf输出都是10.&a和p是一伙的printf输出是一样的。（p是数，p是地址）
题目：交换数据
void swap(int *x, int *y) {
int temp = *x;
*x = *y;
*y = temp;
}

int main() {
int a = 3, b = 5;
printf(“交换前：a = %d, b = %d\n”, a, b);
swap(&a, &b);
printf(“交换后：a = %d, b = %d\n”, a, b);
return 0;
}
这里用到了函数的知识。定义的函数中先把x指向的值存起来，然后y的指向的值赋x的指向的方向，最后把x指向的值赋给y指向的地方。
太深奥了，ai给我解释好久才明白。。
题目：遍历数组
int *p = arr; // 让 p 指向数组的第一个也就是arr[0]
for (int i = 0; i < n; i++) {
scanf(“%d”, p); // 这里直接写 p，因为 p 本身就是地址，不需要加 &
printf(“%d\n”, *p); // *p 就是取出 p 当前指向的那个格子里的值
p++; // 让指针往后跑
}
当然我看到题第一反应是直接用数组学的东西来写
for (int i = 0; i < n; i++) {
scanf(“%d”, &arr[i]); // 读入存到第 i 个格子
printf(“%d\n”, arr[i]); // 打印第 i 个格子的内容
}
指针的方法反正是不用管那么多i了，每次循环让指针（p）往后跑一步。
ai给我一个更nb的
int main() {
int n;
scanf(“%d”, &n);
int arr[n];
// 关键点1：定义指针 p，指向数组第一个元素
int *p = arr;
// 关键点2：用指针读入数据（不含下标变量）
// 循环条件用 p < arr + n，意思是：只要指针还没走到数组末尾，就继续
while (p < arr + n) {
// p 本身就是地址，直接用，不用加 &
scanf(“%d”, p);
p++; // 指针往后挪一格，指向下一个格子
}
// 关键点3：重置指针，重新从头开始遍历
p = arr; // 前面循环结束，p 已经跑到末尾了，必须让它重新指回开头
// 用指针遍历输出（不含下标变量）
while (p < arr + n) {
printf(“%d\n”, *p); // *p 取出当前指向的值
p++; // 继续往后挪
}
return 0;
}
比刚才那个牛在 arr+n 不是简单加法，是和指针一样代表位置后移，还有重置遍历的操作。
